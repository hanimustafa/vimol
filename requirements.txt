# vimol runtime dependencies.
#
# Install the package itself with `pip install vimol` (or `pip install -e .`
# from a checkout); this file pins the same requirements for reproducible
# environments and CI. Everything else vimol uses is standard library.
#
# moderngl is the OpenGL backend (~120 KB of wheels). It is installed
# unconditionally; at runtime vimol falls back to its pure-numpy CPU
# raycaster wherever no OpenGL context can be created.
#
# numba JIT-compiles that CPU raycaster's inner loop (10-25x faster,
# bit-identical output). Installed unconditionally too; render.py still
# tolerates it being missing/uncompilable at runtime as a defensive
# fallback, not because it's meant to be left out.
numpy>=1.20
moderngl>=5.6
numba>=0.59

# Optional extras (not installed by this file):
#   test suite          pip install "vimol[test]"  ->  pytest>=7
